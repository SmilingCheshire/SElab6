# The vital_signs_analyzer script implements the following requirements:
# 1/ The system should accept heart rate (HR) and blood oxygen saturation (SpO2) readings. The values should be normalized into a 0-1 range using defined physiological limits.
# 2/ The system should detect the patient's status based on thresholds for normalized HR and SpO2.
# 3/ The user should be able to override the status with a manually set value.
# 4/ The system should support displaying the status (if enabled) and logging it to a file (if enabled).
HR_MIN = 40
HR_MAX = 180
SPO2_MIN = 0
SPO2_MAX = 100

debug_mode = False
last_values = {}


def normalize_hr(hr: float) -> float:
    return (hr - HR_MIN) / (HR_MAX - HR_MIN)


def normalize_spo2(spo2: float) -> float:
    return spo2 / SPO2_MAX


def analyze(hr: float, spo2: float, log: bool = True, override_status: str | None = None, verbose: int = 0) -> str:
    last_values["hr"] = hr
    last_values["spo2"] = spo2

    if verbose:
        print("Analyzing vital data:", hr, spo2)

    n_hr = normalize_hr(hr)
    n_spo2 = normalize_spo2(spo2)

    if override_status is not None:
        status = override_status
    else:
        if n_hr > 0.7 and n_spo2 < 0.3:
            status = "CRITICAL"
        elif n_hr > 0.4:
            if spo2 < 88 or hr > 200:
                status = "SEVERE"
            else:
                status = "WARNING"
        else:
            status = "OK"

    if log:
        try:
            with open("vitals_log.txt", "a") as f:
                f.write(
                    f"HR={hr}, SpO2={spo2}, "
                    f"N_HR={n_hr:.2f}, N_SPO2={n_spo2:.2f}, "
                    f"STATUS={status}\n"
                )
        except Exception as e:
            print("Logging failed:", e)

    return status


if __name__ == "__main__":
    hr = 999
    spo2 = -3

    result = analyze(hr, spo2, log=True, override_status=None, verbose=1)

    print("Finished analysis, status:", result)
    print("Last values:", last_values)
